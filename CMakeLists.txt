cmake_minimum_required(VERSION 3.25)
project(meta-cue VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(QT_MAJOR_VERSION 6)
include(KDEInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui GuiPrivate Quick Qml DBus Test Widgets OpenGL)
find_package(KF6 REQUIRED COMPONENTS Config CoreAddons GlobalAccel)
find_package(KGlobalAccelD REQUIRED)
find_package(KWin REQUIRED)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
add_library(meta_cue_input MODULE src/inputplugin.cpp)
target_link_libraries(meta_cue_input PRIVATE KWin::kwin Qt6::DBus KF6::ConfigCore KF6::CoreAddons K::KGlobalAccelD)
target_include_directories(meta_cue_input PRIVATE src)
set_target_properties(meta_cue_input PROPERTIES PREFIX "")
add_library(metacuebackend MODULE src/backend.cpp)
target_link_libraries(metacuebackend PRIVATE Qt6::Quick Qt6::Qml Qt6::DBus Plasma::Plasma Plasma::PlasmaQuick KF6::ConfigCore)
set_target_properties(metacuebackend PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/contents/ui/backend")
file(COPY package/ DESTINATION "${CMAKE_BINARY_DIR}/package")
enable_testing()
add_executable(test_gesture tests/test_gesture.cpp)
target_include_directories(test_gesture PRIVATE src)
target_link_libraries(test_gesture PRIVATE Qt6::Test)
add_test(NAME gesture COMMAND test_gesture)
# Isolated compositor integration tests; these targets are never installed.
add_library(meta_cue_test_driver MODULE tests/input_driver.cpp)
target_link_libraries(meta_cue_test_driver PRIVATE KWin::kwin Qt6::DBus)
set_target_properties(meta_cue_test_driver PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/kwin/plugins")
add_executable(test_integration tests/test_integration.cpp)
target_link_libraries(test_integration PRIVATE Qt6::Test Qt6::DBus Qt6::Gui KF6::GlobalAccel)
